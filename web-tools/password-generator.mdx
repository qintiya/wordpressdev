Strong Password Generator membantu Anda membuat kata sandi yang kuat dan sulit ditebak langsung di browser. Semua proses pembuatan password berjalan secara lokal di perangkat Anda: tidak ada data yang dikirim ke server.

Kekuatan password diukur dari entropi (bit acak). Entropi lebih tinggi berarti password lebih sulit ditebak dengan serangan brute force. Alat ini menggunakan `crypto.getRandomValues` untuk menghasilkan angka acak kriptografis dan menjamin setidaknya satu karakter dari setiap kategori yang dipilih.

## Cara menggunakan

<Steps>
  <Step title="Atur panjang password">
    Geser slider untuk memilih panjang password antara 8 hingga 64 karakter. Semakin panjang, semakin tinggi entropi.
  </Step>
  <Step title="Pilih jenis karakter">
    Centang setidaknya satu kategori: huruf besar, huruf kecil, angka, atau simbol. Anda juga dapat mengecualikan karakter ambigu (Il1O0) agar password lebih mudah dibaca.
  </Step>
  <Step title="Klik Generate">
    Tekan tombol Generate untuk membuat password baru. Password akan dihasilkan secara otomatis saat halaman dimuat.
  </Step>
  <Step title="Salin ke clipboard">
    Gunakan tombol Copy untuk menyalin password ke clipboard dengan aman.
  </Step>
</Steps>

## FAQ

<AccordionGroup>
  <Accordion title="Apakah password yang dihasilkan aman?">
    Ya. Password dibuat menggunakan `crypto.getRandomValues` yang merupakan sumber acak kriptografis dari browser. Tidak ada pola yang dapat diprediksi.
  </Accordion>

  <Accordion title="Apakah password saya disimpan di server?">
    Tidak. Semua pembuatan password terjadi sepenuhnya di browser Anda. Tidak ada data yang dikirim ke server AWDEV.
  </Accordion>

  <Accordion title="Mengapa entropi penting?">
    Entropi mengukur seberapa tidak terduga sebuah password. Entropi tinggi berarti ruang kemungkinan password sangat besar, sehingga serangan brute force membutuhkan waktu yang sangat lama.
  </Accordion>

  <Accordion title="Apa yang dimaksud karakter ambigu?">
    Karakter seperti huruf I besar, huruf l kecil, angka 1, huruf O besar, dan angka 0 sering tertukar saat dibaca. Mengecualikan karakter ini membuat password lebih mudah disalin dengan benar.
  </Accordion>
</AccordionGroup>

export const PasswordGenerator = () => {
  const [length, setLength] = useState(16);
  const [useUpper, setUseUpper] = useState(true);
  const [useLower, setUseLower] = useState(true);
  const [useNumbers, setUseNumbers] = useState(true);
  const [useSymbols, setUseSymbols] = useState(true);
  const [excludeAmbiguous, setExcludeAmbiguous] = useState(true);
  const [password, setPassword] = useState("");
  const [copied, setCopied] = useState(false);

  const UPPER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
  const LOWER = "abcdefghijklmnopqrstuvwxyz";
  const NUMBERS = "0123456789";
  const SYMBOLS = "!@#$%^&*()_+-=[]{}|;:,.<>?";
  const AMBIGUOUS = "Il1O0";

  const getPool = () => {
    let pool = "";
    if (useUpper) pool += UPPER;
    if (useLower) pool += LOWER;
    if (useNumbers) pool += NUMBERS;
    if (useSymbols) pool += SYMBOLS;
    if (excludeAmbiguous) {
      pool = pool.split("").filter(c => !AMBIGUOUS.includes(c)).join("");
    }
    return pool;
  };

  const randomInt = (max) => {
    const arr = new Uint32Array(1);
    const threshold = (0x100000000 - (0x100000000 % max));
    let val;
    do {
      window.crypto.getRandomValues(arr);
      val = arr[0];
    } while (val >= threshold);
    return val % max;
  };

  const shuffle = (arr) => {
    for (let i = arr.length - 1; i > 0; i--) {
      const j = randomInt(i + 1);
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr;
  };

  const generate = () => {
    const pool = getPool();
    if (pool.length === 0) {
      setPassword("");
      return;
    }

    const required = [];
    if (useUpper) {
      let p = UPPER;
      if (excludeAmbiguous) p = p.split("").filter(c => !AMBIGUOUS.includes(c)).join("");
      if (p.length) required.push(p[randomInt(p.length)]);
    }
    if (useLower) {
      let p = LOWER;
      if (excludeAmbiguous) p = p.split("").filter(c => !AMBIGUOUS.includes(c)).join("");
      if (p.length) required.push(p[randomInt(p.length)]);
    }
    if (useNumbers) {
      let p = NUMBERS;
      if (excludeAmbiguous) p = p.split("").filter(c => !AMBIGUOUS.includes(c)).join("");
      if (p.length) required.push(p[randomInt(p.length)]);
    }
    if (useSymbols) {
      let p = SYMBOLS;
      if (excludeAmbiguous) p = p.split("").filter(c => !AMBIGUOUS.includes(c)).join("");
      if (p.length) required.push(p[randomInt(p.length)]);
    }

    const result = [...required];
    while (result.length < length) {
      result.push(pool[randomInt(pool.length)]);
    }
    const shuffled = shuffle(result);
    setPassword(shuffled.join(""));
  };

  useEffect(() => {
    generate();
  }, []);

  useEffect(() => {
    generate();
  }, [length, useUpper, useLower, useNumbers, useSymbols, excludeAmbiguous]);

  const pool = getPool();
  const poolSize = pool.length;
  const entropy = password.length > 0 && poolSize > 0 ? password.length * Math.log2(poolSize) : 0;
  let strengthLabel = "Pilih setidaknya satu kategori";
  let strengthColor = "#dc2626";
  if (poolSize > 0) {
    if (entropy < 50) { strengthLabel = "Weak"; strengthColor = "#dc2626"; }
    else if (entropy < 80) { strengthLabel = "Medium"; strengthColor = "#d97706"; }
    else if (entropy < 110) { strengthLabel = "Strong"; strengthColor = "#315c45"; }
    else { strengthLabel = "Very strong"; strengthColor = "#4f9a72"; }
  }

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(password);
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    } catch (e) {}
  };

  return (
    <div className="not-prose" style={{ display: "flex", flexDirection: "column", gap: 16 }}>
      <div style={{ display: "flex", gap: 8, flexWrap: "wrap", alignItems: "center" }}>
        <button type="button" onClick={generate} style={{ padding: "8px 14px", borderRadius: 8, border: "1px solid rgba(127,127,127,.35)", background: "transparent", color: "inherit", cursor: "pointer" }}>
          Generate
        </button>
        <button type="button" onClick={handleCopy} style={{ padding: "8px 14px", borderRadius: 8, border: "1px solid rgba(127,127,127,.35)", background: "transparent", color: "inherit", cursor: "pointer" }}>
          {copied ? "Copied!" : "Copy"}
        </button>
      </div>

      <input
        type="text"
        readOnly
        value={password}
        style={{ background: "transparent", color: "inherit", border: "1px solid rgba(127,127,127,.35)", borderRadius: 8, padding: 12, width: "100%", font: "inherit", fontFamily: "monospace" }}
      />

      <div style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 8, padding: 12 }}>
        <div style={{ fontWeight: 700, marginBottom: 8 }}>Kekuatan Password</div>
        <div style={{ display: "flex", alignItems: "center", gap: 12, flexWrap: "wrap" }}>
          <div style={{ flex: 1, minWidth: 120, height: 8, background: "rgba(127,127,127,.2)", borderRadius: 4, overflow: "hidden" }}>
            <div style={{ width: poolSize > 0 ? Math.min(100, (entropy / 150) * 100) + "%" : "0%", height: "100%", background: strengthColor, borderRadius: 4 }} />
          </div>
          <div style={{ fontWeight: 700, color: strengthColor, minWidth: 120 }}>{strengthLabel}</div>
        </div>
        <div style={{ marginTop: 6, fontSize: 12, opacity: 0.8 }}>Entropi: {entropy.toFixed(1)} bit</div>
      </div>

      <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(220px, 1fr))", gap: 12 }}>
        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="range" min={8} max={64} value={length} onChange={(e) => setLength(Number(e.target.value))} style={{ flex: 1 }} />
          <span style={{ minWidth: 28 }}>{length}</span>
        </label>

        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="checkbox" checked={useUpper} onChange={(e) => setUseUpper(e.target.checked)} />
          Huruf besar (A-Z)
        </label>

        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="checkbox" checked={useLower} onChange={(e) => setUseLower(e.target.checked)} />
          Huruf kecil (a-z)
        </label>

        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="checkbox" checked={useNumbers} onChange={(e) => setUseNumbers(e.target.checked)} />
          Angka (0-9)
        </label>

        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="checkbox" checked={useSymbols} onChange={(e) => setUseSymbols(e.target.checked)} />
          Simbol (!@#$...)
        </label>

        <label style={{ display: "flex", alignItems: "center", gap: 8, cursor: "pointer" }}>
          <input type="checkbox" checked={excludeAmbiguous} onChange={(e) => setExcludeAmbiguous(e.target.checked)} />
          Hindari ambigu (Il1O0)
        </label>
      </div>
    </div>
  );
};
